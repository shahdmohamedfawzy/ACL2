// EXCEEDS EXPECTATIONS RULE - Solo Female Travellers
// A hotel exceeds expectations when:
// (cleanliness_base + comfort_base + facilities_base) >= 
// AVG(score_cleanliness + score_comfort + score_facilities) for the demographic

MATCH (t:Traveller {type: "Solo", gender: "Female"})-[:WROTE]->(r:Review)-[:REVIEWED]->(h:Hotel)
WITH t, h, r, t.age AS age_group

// Calculate average scores from reviews for this demographic
WITH h, age_group, 
     avg(r.score_cleanliness) AS avg_cleanliness,
     avg(r.score_comfort) AS avg_comfort,
     avg(r.score_facilities) AS avg_facilities,
     h.cleanliness_base AS cleanliness_base,
     h.comfort_base AS comfort_base,
     h.facilities_base AS facilities_base

// Calculate base sum and review average
WITH h, age_group,
     cleanliness_base + comfort_base + facilities_base AS base_sum,
     avg_cleanliness + avg_comfort + avg_facilities AS review_avg

// Filter only hotels that exceed expectations
WHERE base_sum >= review_avg

// Calculate improvement percentage
WITH age_group, 
     ((base_sum - review_avg) / review_avg) * 100 AS improvement_percentage

// Group by age group and calculate statistics
WITH age_group,
     min(improvement_percentage) AS min_improve,
     max(improvement_percentage) AS max_improve,
     avg(improvement_percentage) AS avg_improve

RETURN age_group, 
       round(min_improve, 2) AS min_improve,
       round(max_improve, 2) AS max_improve,
       round(avg_improve, 2) AS avg_improve
ORDER BY age_group
